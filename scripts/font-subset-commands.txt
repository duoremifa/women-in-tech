# 字体子集化命令示例
# 需要安装 pyftsubset (fonttools)

# Inter 字体子集化
pyftsubset fonts/Inter-Regular.ttf \
  --unicodes-file=public/fonts/charset.txt \
  --output-file=public/fonts/inter-subset.woff2 \
  --flavor=woff2 \
  --layout-features='*' \
  --glyph-names \
  --symbol-cmap \
  --name-IDs='*'
